STEP 6. RUN THE MAPREDUCE JOBS (Hadoop Streaming) - Sections 8, 10
--------------------------------------------------------------------------------
export STREAM_JAR=$HADOOP_HOME/share/hadoop/tools/lib/hadoop-streaming-*.jar
export IN=/taxi_project/input/cleaned

# (a) Hourly Taxi Demand
hadoop jar $STREAM_JAR \
  -input $IN -output /taxi_project/output/hourly \
  -mapper "python3 mapper_hourly.py" -reducer "python3 reducer_hourly.py" \
  -file mappers/mapper_hourly.py -file reducers/reducer_hourly.py

# (b) Daily Demand
hadoop jar $STREAM_JAR \
  -input $IN -output /taxi_project/output/daily \
  -mapper "python3 mapper_daily.py" -reducer "python3 reducer_daily.py" \
  -file mappers/mapper_daily.py -file reducers/reducer_daily.py

# (c) Pickup Location Analysis
hadoop jar $STREAM_JAR \
  -input $IN -output /taxi_project/output/locations \
  -mapper "python3 mapper_location.py" -reducer "python3 reducer_location.py" \
  -file mappers/mapper_location.py -file reducers/reducer_location.py

# (d) Revenue by Pickup Location  <-- STAGE 1 of the compulsory 2-stage job
hadoop jar $STREAM_JAR \
  -input $IN -output /taxi_project/output/revenue \

  -file mappers/mapper_revenue.py -file reducers/reducer_revenue.py

--------------------------------------------------------------------------------
STEP 7. MULTI-STAGE MAPREDUCE - COMPULSORY (Section 9)
Job 2 consumes Job 1's HDFS output (/taxi_project/output/revenue) and
produces the Top 10 highest-revenue pickup zones.
reducer_revenue.py output columns: PU  count  fare_sum  tip_sum  revenue_sum  avg_fare  avg_dist
                                     0    1       2         3         4          5          6
=> sort field for "revenue_sum" is index 3 (0=key(PU), 1=count, 2=fare_sum, 3=tip_sum...)
NOTE: recount carefully against your actual reducer output before running --
field indices are documented at the top of reducer_topn.py and match the
tab-separated VALUE columns that follow the key.
--------------------------------------------------------------------------------
# First show the INTERMEDIATE Stage-1 HDFS output (required evidence):
hdfs dfs -cat /taxi_project/output/revenue/part-* | head -20
hdfs dfs -ls -h /taxi_project/output/revenue

# Stage 2: Top 10 pickup zones by total revenue (revenue_sum is field index 3)
hadoop jar $STREAM_JAR \
  -input /taxi_project/output/revenue \
  -output /taxi_project/output/revenue_top10 \
  -mapper "python3 mapper_topn.py" \
  -reducer "python3 reducer_topn.py 3 10 desc" \
  -file mappers/mapper_topn.py -file reducers/reducer_topn.py \
  -numReduceTasks 1

hdfs dfs -cat /taxi_project/output/revenue_top10/part-*


STEP 8. REUSE THE SAME TOP-N PATTERN FOR OTHER REQUIRED TOP/BOTTOM LISTS
--------------------------------------------------------------------------------
# Top 10 pickup zones by TRIP COUNT (reducer_location.py output: zone<TAB>count, count=field idx 1)
hadoop jar $STREAM_JAR \
  -input /taxi_project/output/locations \
  -output /taxi_project/output/top10_locations \
  -mapper "python3 mapper_topn.py" -reducer "python3 reducer_topn.py 1 10 desc" \
  -file mappers/mapper_topn.py -file reducers/reducer_topn.py -numReduceTasks 1

# Bottom 10 pickup zones by TRIP COUNT
hadoop jar $STREAM_JAR \
  -input /taxi_project/output/locations \
  -output /taxi_project/output/bottom10_locations \
  -mapper "python3 mapper_topn.py" -reducer "python3 reducer_topn.py 1 10 asc" \
  -file mappers/mapper_topn.py -file reducers/reducer_topn.py -numReduceTasks 1

# Top 20 routes by trip COUNT (reducer_route.py output: route<TAB>count,fare_sum,revenue_sum; count=idx1)
hadoop jar $STREAM_JAR \
  -input /taxi_project/output/routes \
  -output /taxi_project/output/routes_top20_count \
  -mapper "python3 mapper_topn.py" -reducer "python3 reducer_topn.py 1 20 desc" \
  -file mappers/mapper_topn.py -file reducers/reducer_topn.py -numReduceTasks 1

# Top 20 routes by REVENUE (revenue_sum = idx3)
hadoop jar $STREAM_JAR \
  -input /taxi_project/output/routes \
  -output /taxi_project/output/routes_top20_revenue \
  -mapper "python3 mapper_topn.py" -reducer "python3 reducer_topn.py 3 20 desc" \
  -file mappers/mapper_topn.py -file reducers/reducer_topn.py -numReduceTasks 1
